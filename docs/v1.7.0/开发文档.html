3. 环境搭建

3.1 系统要求

环境要求
Node.js18+ LTS
操作系统Windows 10+ / Linux (Ubuntu 20.04+推荐) / macOS
浏览器Chrome 90+ / Edge 90+ / Firefox 90+

3.2 本地开发启动

# 1. 克隆仓库
git clone git@github.com:qubin2025/doc-management-system.git
cd doc-management-system

# 2. 安装前端依赖
npm install

# 3. 安装后端依赖
cd backend && npm install && cd ..

# 4. 配置环境变量
cp .env.example .env          # 编辑前端环境
cp backend/.env.example backend/.env  # 编辑后端环境（填入API Keys）

# 5. 启动开发服务器
npm run dev                   # 前端 localhost:5173
cd backend && node server.js  # 后端 localhost:3000

3.3 生产构建

npm run build     # 前端构建 → dist/
cd backend && node server.js  # 启动后端

5. 数据库设计

-- projects 项目表
CREATE TABLE projects (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  name TEXT NOT NULL UNIQUE,
  created_at TEXT NOT NULL DEFAULT (datetime('now'))
);

-- documents 文档表
CREATE TABLE documents (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  project_id INTEGER NOT NULL REFERENCES projects(id),
  doc_id TEXT NOT NULL,
  file_name TEXT NOT NULL,
  file_data TEXT,
  file_path TEXT,
  upload_time TEXT,
  uploader TEXT,
  version TEXT DEFAULT 'V1.0',
  standard TEXT DEFAULT 'DB11/T695-2025'
);
CREATE INDEX idx_documents_project ON documents(project_id);
CREATE INDEX idx_documents_doc_id ON documents(doc_id);
CREATE INDEX idx_documents_standard ON documents(standard);

-- users 用户表
CREATE TABLE users (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  username TEXT NOT NULL UNIQUE,
  password_hash TEXT NOT NULL,
  display_name TEXT DEFAULT '',
  role TEXT CHECK(role IN ('admin','project_manager','construction_unit','viewer')),
  permissions TEXT DEFAULT '{}',
  is_active INTEGER DEFAULT 1,
  created_at TEXT DEFAULT (datetime('now'))
);

-- sessions 会话表
CREATE TABLE sessions (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  token TEXT NOT NULL UNIQUE,
  expires_at TEXT NOT NULL,
  created_at TEXT NOT NULL DEFAULT (datetime('now'))
);
CREATE INDEX idx_sessions_token ON sessions(token);

数据库文件路径：backend/data/planning.db（WAL模式），支持自动Schema迁移和种子数据插入。
