# 生产服务器 crontab 配置
# 安装: crontab deploy/crontab.txt
# 查看: crontab -l
# 编辑: crontab -e

# 数据库备份 — 每天凌晨2点执行
0 2 * * * /opt/doc-mgmt/scripts/backup.sh >> /var/log/doc-mgmt-backup.log 2>&1

# 数据清理（过期会话）— 每周日凌晨3点
0 3 * * 0 node /opt/doc-mgmt/scripts/clean-test-data.js >> /var/log/doc-mgmt-cleanup.log 2>&1

# 日志轮转 — 每天凌晨4点（配合logrotate）
0 4 * * * /usr/sbin/logrotate -s /var/lib/logrotate.status /etc/logrotate.d/doc-mgmt
